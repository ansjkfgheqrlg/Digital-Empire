\#\# CLASSIFICAZIONE PROSPECT — SISTEMA TIER  
   
Quando ricevi una tabella, per ogni prospect assegni un Tier prima di scrivere qualsiasi cosa.  
   
\`\`\`  
TIER A1 — WEB DESIGN (priorità massima)  
────────────────────────────────────────  
Segnali:  
→ Ads attive ma nessuna landing page (solo profilo IG)  
→ Landing su dominio di terzo livello o builder generico  
→ Landing non coerente col posizionamento del brand  
→ "Prenota ora" / "Scopri di più" che porta al feed IG  
→ Linktree o link.bio come unica destinazione  
→ Sito assente o vetrina senza funnel  
   
Angolo messaggio: stai pagando per portare  
persone in un posto che non converte.  
Il budget esce, i contatti non entrano.  
   
   
TIER A2 — CRO (priorità alta)  
───────────────────────────────  
Segnali:  
→ Sito o landing page presente e curata  
→ Ads attive (visibili da Meta Ad Library)  
→ Form lungo come primo punto di contatto  
→ CTA confusa o assente above the fold  
→ Nessuna social proof visibile  
→ Headline generica (potrebbe essere di chiunque)  
→ Funnel con step mancanti o checkout con friction  
   
Angolo messaggio: il traffico arriva,  
ma il funnel perde persone prima che convertano.  
   
   
TIER B — POTENZIALE (seconda priorità)  
────────────────────────────────────────  
Segnali:  
→ Audience grande (100k+) ma senza sistema di gestione  
→ Problema presente ma meno critico o urgente  
→ ICP parzialmente soddisfatto  
   
Angolo messaggio: stai lasciando sul tavolo  
conversioni che il tuo volume di traffico  
dovrebbe già generare.  
   
   
TIER C — FUORI ICP (non scrivere)  
───────────────────────────────────  
Segnali:  
→ Nessuna ads attiva (non stanno investendo)  
→ Nessun prodotto/servizio con prezzo chiaro  
→ Business locale senza presenza digitale rilevante  
→ Già strutturati con team marketing interno evidente  
   
Azione: segnali al founder perché è fuori target.  
Non scrivi nessun messaggio.

\#\# BRAND VOICE — REGOLE NON NEGOZIABILI  
   
Il tono è quello di \*\*un esperto che parla a un collega\*\* — non di un venditore, non di un consulente formale, non di un guru motivazionale.  
   
\#\#\# Regole Assolute  
   
\`\`\`  
✅ USA SEMPRE:  
→ "tu" — mai "lei", mai "voi" generico  
→ Frasi brevi. Max 15-20 parole per frase  
→ Paragrafi max 2-3 righe  
→ Linguaggio parlato, non scritto formale  
→ Immagini concrete ("sparisce", "brucia", "esce")  
→ Imperfezioni naturali del parlato scritto  
→ Zero punto finale nei DM (nessuno lo mette)  
   
❌ NON USARE MAI:  
→ "ottimizzare" / "implementare" / "strategia"  
→ "opportunità" / "potenziale" / "soluzioni"  
→ "mi piacerebbe proporre" / "sarei lieto di"  
→ "ho notato che potresti beneficiare"  
→ Complimenti vuoti ("ottimo brand\!", "bel profilo\!")  
→ Parole da guru: "mindset", "game changer",  
  "trasforma", "sblocca", "rivoluzionario"  
→ Struttura troppo pulita e simmetrica  
  (si vede che è costruita)  
→ Qualsiasi frase che potrebbe stare  
  in un template inviato a 500 persone  
\`\`\`

"""  
prospect\_classifier.py  
Classifica prospect Instagram per Digital Empire Outreach  
"""  
   
from dataclasses import dataclass  
from typing import Optional  
from enum import Enum  
   
   
class Tier(Enum):  
   A1\_WEBDESIGN \= "A1 — Web Design"  
   A2\_CRO \= "A2 — CRO"  
   B\_POTENZIALE \= "B — Potenziale"  
   C\_FUORI\_ICP \= "C — Fuori ICP"  
   
   
class Servizio(Enum):  
   WEB\_DESIGN \= "Landing Page / Web Design"  
   CRO \= "Conversion Rate Optimization"  
   NESSUNO \= "Fuori ICP — non contattare"  
   
   
@dataclass  
class Prospect:  
   instagram\_url: str  
   nome: str  
   azienda: str  
   website: Optional\[str\]  
   service\_need: str  
   note\_opportunita: str  
   
   \# Segnali rilevati  
   ha\_landing\_page: bool \= False  
   ha\_ads\_attive: bool \= False  
   landing\_su\_dominio\_terzo: bool \= False  
   landing\_incoerente\_brand: bool \= False  
   cta\_porta\_a\_ig: bool \= False  
   ha\_form\_lungo: bool \= False  
   manca\_social\_proof: bool \= False  
   headline\_generica: bool \= False  
   audience\_grande: bool \= False  \# 100k+ follower  
   
   def classifica(self) \-\> tuple\[Tier, Servizio, str\]:  
       """  
       Restituisce (Tier, Servizio, problema\_core)  
       """  
       \# TIER C — Fuori ICP  
       if not self.ha\_ads\_attive:  
           return (  
               Tier.C\_FUORI\_ICP,  
               Servizio.NESSUNO,  
               "Nessuna ads attiva — non sta investendo"  
           )  
   
       \# TIER A1 — Web Design  
       if (  
           not self.ha\_landing\_page  
           or self.cta\_porta\_a\_ig  
           or self.landing\_su\_dominio\_terzo  
           or self.landing\_incoerente\_brand  
       ):  
           problemi \= \[\]  
           if not self.ha\_landing\_page or self.cta\_porta\_a\_ig:  
               problemi.append(  
                   "ads attive che portano al profilo IG — "  
                   "ogni clic pagato sparisce senza catturare nulla"  
               )  
           if self.landing\_su\_dominio\_terzo:  
               problemi.append(  
                   "landing su dominio di terzo livello — "  
                   "brucia la fiducia prima della prima riga"  
               )  
           if self.landing\_incoerente\_brand:  
               problemi.append(  
                   "landing non coerente col posizionamento — "  
                   "disallineamento tra brand percepito e pagina reale"  
               )  
           return (  
               Tier.A1\_WEBDESIGN,  
               Servizio.WEB\_DESIGN,  
               " / ".join(problemi)  
           )  
   
       \# TIER A2 — CRO  
       segnali\_cro \= sum(\[  
           self.ha\_form\_lungo,  
           self.manca\_social\_proof,  
           self.headline\_generica,  
       \])  
       if segnali\_cro \>= 1:  
           problemi \= \[\]  
           if self.ha\_form\_lungo:  
               problemi.append(  
                   "form lungo come primo contatto — "  
                   "dimezza le compilazioni prima ancora di iniziare"  
               )  
           if self.manca\_social\_proof:  
               problemi.append(  
                   "zero social proof visibile — "  
                   "chi arriva non ha motivo di fidarsi"  
               )  
           if self.headline\_generica:  
               problemi.append(  
                   "headline above the fold generica — "  
                   "potrebbe essere di chiunque nel settore"  
               )  
           return (  
               Tier.A2\_CRO,  
               Servizio.CRO,  
               " / ".join(problemi)  
           )  
   
       \# TIER B — Potenziale  
       if self.audience\_grande:  
           return (  
               Tier.B\_POTENZIALE,  
               Servizio.CRO,  
               "audience grande senza sistema di pre-qualificazione lead"  
           )  
   
       \# Default B se ha ads ma non rientra nei casi precedenti  
       return (  
           Tier.B\_POTENZIALE,  
           Servizio.WEB\_DESIGN,  
           "presenza digitale con margini di ottimizzazione non critici"  
       )  
   
   def stampa\_scheda(self):  
       tier, servizio, problema \= self.classifica()  
       print(f"\\n{'━'\*55}")  
       print(f"PROSPECT: {self.instagram\_url}")  
       print(f"NOME:     {self.nome} — {self.azienda}")  
       print(f"TIER:     {tier.value}")  
       print(f"SERVIZIO: {servizio.value}")  
       print(f"PROBLEMA: {problema}")  
       print(f"{'━'\*55}")

   testo: str

def audita\_messaggio(testo: str) \-\> AuditResult:  
   result \= AuditResult(testo=testo)  
   
   \# Penalità per lunghezza eccessiva  
   if result.righe \> 8:  
       result.violazioni.append((  
           "LUNGHEZZA",  
           f"{result.righe} righe — max 8\. Taglia."  
       ))  
       result.score \-= 20  
   
   \# Controlla pattern AI  
   for pattern, descrizione in PATTERN\_AI:  
       matches \= re.findall(pattern, testo, re.IGNORECASE)  
       if matches:  
           result.violazioni.append((  
               "LINGUAGGIO AI",  
               f'"{matches\[0\]}" — {descrizione}'  
           ))  
           result.score \-= 10  
   
   \# Controlla segnali umani  
   for pattern, descrizione in SEGNALI\_UMANI:  
       if re.search(pattern, testo, re.IGNORECASE):  
           result.segnali\_positivi.append(descrizione)  
   
   \# Controlla punto finale (nessuno lo mette nei DM)  
   righe \= \[r.strip() for r in testo.strip().split("\\n") if r.strip()\]  
   for riga in righe:  
       if riga.endswith(".") and len(riga) \> 20:  
           result.violazioni.append((  
               "PUNTO FINALE",  
               f'"{riga\[-30:\]}" — rimuovi il punto finale nei DM'  
           ))  
           result.score \-= 5  
           break  
   
   \# Prima riga deve contenere il problema  
   if righe:  
       prima\_riga \= righe\[0\].lower()  
       segnali\_problema \= \[  
           "dominio", "link", "pagina", "ads", "inserzione",  
           "profilo", "form", "clic", "click", "landing",  
           "brand", "sito", "budget", "stai", "hai", "la tua"  
       \]  
       if not any(s in prima\_riga for s in segnali\_problema):  
           result.violazioni.append((  
               "PRIMA RIGA",  
               "La prima riga non sembra nominare il problema. "  
               "Deve iniziare già dal problema specifico."  
           ))  
           result.score \-= 15  
   
   result.score \= max(0, result.score)  
   return result
