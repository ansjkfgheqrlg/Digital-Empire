    def __init__(self):
        self.state = "IDLE"

    def validate_input(self, user_input):
        """PHASE 1: Strict Variable Validation"""

        missing = [key for key in required_keys if key not in user_input]

            self.state = "AWAITING_INPUT"
            return f"MISSING_VARIABLES: {missing}"
        
        self.state = "ANALYZING"
        return "INPUT_VALID"

    def select_strategy(self, audience_awareness, price_point):
        """PHASE 2: Algorithmic Strategy Selection"""
        if audience_awareness == "UNAWAWARE" and price_point == "HIGH":
            return "CONTENT_EDUCATION_FUNNEL"
        elif audience_awareness == "PROBLEM_AWARE" and price_point == "MID":
            return "LEAD_MAGNET_SEQUENCE"
        elif audience_awareness == "SOLUTION_AWARE" and price_point == "LOW":
            return "DIRECT_RESPONSE_VSL"
        else:
            return "HYBRID_AUTHORITY_SYSTEM"

    def quality_gate_check(self, draft_output):
        """PHASE 3: Pre-Output Validation"""
        errors = []

            errors.append("MISSING_RETENTION_LOOP")
