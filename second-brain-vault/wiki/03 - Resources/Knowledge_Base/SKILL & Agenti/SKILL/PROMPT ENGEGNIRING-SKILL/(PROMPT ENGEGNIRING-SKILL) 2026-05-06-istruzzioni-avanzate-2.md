<philosophical_foundation> 
   
  <axioms> 
    <!-- Fundamental truths that cannot be violated --> 
     
    <axiom id="A1" name="Structure Supremacy" weight="ABSOLUTE"> 
      <statement>Natural language is inherently lossy; structured syntax is 
lossless</statement> 
      <implication>Always prefer structured formats (XML, Markdown headers, delimiters) 
over flowing prose</implication> 
      <evidence>Structured prompts reduce interpretation variance by 60-80%</evidence> 
    </axiom> 
     
    <axiom id="A2" name="Explicit Dominance" weight="ABSOLUTE"> 
      <statement>Never assume an LLM will correctly infer anything left unstated</statement> 
      <implication>State everything explicitly; implicit expectations are failure 
points</implication> 
      <evidence>Explicit instructions outperform implicit ones by 45% on complex 
tasks</evidence> 
    </axiom> 
     
    <axiom id="A3" name="Constraint Power" weight="ABSOLUTE"> 
      <statement>Defining what NOT to do is equally important as defining what TO 
do</statement> 
      <implication>Every prompt must include negative constraints to bound the solution 
space</implication> 
      <evidence>Negative constraints reduce off-target outputs by 35-50%</evidence> 
    </axiom> 
     
    <axiom id="A4" name="Reasoning Transparency" weight="ABSOLUTE"> 
      <statement>Forcing visible reasoning paths dramatically reduces errors and 
hallucinations</statement> 
      <implication>Complex tasks require mandatory Chain-of-Thought 
mechanisms</implication> 
      <evidence>CoT reduces hallucinations by 40-60% on reasoning tasks</evidence> 
    </axiom> 

     
    <axiom id="A5" name="Example Superiority" weight="ABSOLUTE"> 
      <statement>One well-crafted example teaches more than extensive theoretical 
instructions</statement> 
      <implication>Include few-shot examples for any task requiring specific 
patterns</implication> 
      <evidence>Few-shot prompting improves format adherence by 70%+</evidence> 
    </axiom> 
     
  </axioms> 
   
  <principles> 
    <!-- Derived principles that guide decision-making --> 
     
    <principle_group name="Structural Principles"> 
      <principle id="P1">MODULARITY: Prompts are assemblies of discrete, reusable 
components</principle> 
      <principle id="P2">HIERARCHY: Information should flow from general to 
specific</principle> 
      <principle id="P3">SEPARATION: Different types of information require different 
containers</principle> 
      <principle id="P4">CONSISTENCY: Same type of information should use same 
formatting throughout</principle> 
    </principle_group> 
     
    <principle_group name="Cognitive Principles"> 
      <principle id="P5">CHUNKING: Break complex instructions into digestible 
segments</principle> 
      <principle id="P6">SEQUENCING: Order instructions in the sequence they should be 
executed</principle> 
      <principle id="P7">EMPHASIS: Critical instructions need structural emphasis (not just 
bold text)</principle> 
      <principle id="P8">REPETITION: Key constraints should appear in multiple sections for 
reinforcement</principle> 
    </principle_group> 
     
    <principle_group name="Optimization Principles"> 
      <principle id="P9">ANTICIPATION: Predict and address likely failure modes

      <principle id="P10">ANCHORING: Provide output examples to anchor expected 
format</principle> 
      <principle id="P11">BOUNDING: Define clear boundaries for scope, length, and 
style</principle> 
      <principle id="P12">VALIDATION: Include success criteria so output can be

    </principle_group> 
     
    <principle_group name="Meta Principles"> 

      <principle id="P13">ITERATION: First drafts are starting points; critique and

      <principle id="P14">ADAPTATION: Prompts should be adjustable for different

      <principle id="P15">DOCUMENTATION: Complex prompts need usage notes for future

      <principle id="P16">VERSIONING: Track changes and maintain prompt evolution

    </principle_group> 
     
  </principles> 
   
  <anti_patterns> 
    <!-- What to actively avoid --> 
     
    <anti_pattern id="AP1" name="Wall of Text" severity="CRITICAL"> 
      <description>Long paragraphs of instructions without structural separation</description> 
      <problem>LLMs lose track of instructions buried in prose</problem> 
      <solution>Use headers, lists, XML tags to separate components</solution> 
    </anti_pattern> 
     
    <anti_pattern id="AP2" name="Vague Quantifiers" severity="HIGH"> 
      <description>Using words like "some", "few", "detailed", "comprehensive"</description> 
      <problem>These words have no objective meaning to an LLM</problem> 
      <solution>Use specific numbers, word counts, or structural requirements</solution> 
    </anti_pattern> 
     
    <anti_pattern id="AP3" name="Orphan Pronouns" severity="HIGH"> 
      <description>Pronouns without clear referents (it, this, that, they)</description> 
      <problem>Ambiguous reference causes incorrect interpretations</problem> 
      <solution>Repeat the noun or use explicit references</solution> 
    </anti_pattern> 
     
    <anti_pattern id="AP4" name="Instruction Burial" severity="MEDIUM"> 
      <description>Critical instructions placed in the middle of long sections</description> 
      <problem>Primacy and recency effects mean middle content gets less weight</problem> 
      <solution>Place critical instructions at start or end; use emphasis</solution> 
    </anti_pattern> 
     
    <anti_pattern id="AP5" name="Conflicting Directives" severity="CRITICAL"> 
      <description>Instructions that contradict each other</description> 
      <problem>LLM must choose; choice may not match intent</problem> 
      <solution>Review for logical consistency; prioritize explicitly if conflicts exist</solution> 
    </anti_pattern> 
     
    <anti_pattern id="AP6" name="Unbounded Output" severity="MEDIUM"> 
      <description>No specification of expected output length or format</description> 
      <problem>LLM defaults may not match expectations</problem> 

      <solution>Always specify format, structure, and approximate length</solution> 
    </anti_pattern> 
     
    <anti_pattern id="AP7" name="Context Starvation" severity="HIGH"> 
      <description>Task instructions without necessary background information</description> 
      <problem>LLM fills gaps with assumptions (often wrong)</problem> 
      <solution>Provide sufficient context even if it seems obvious</solution> 
    </anti_pattern> 
     
    <anti_pattern id="AP8" name="Single-Path Thinking" severity="MEDIUM"> 
      <description>Only one approach offered with no consideration of 
alternatives</description> 
      <problem>Chosen path may not be optimal for all cases</problem> 
      <solution>Consider edge cases; provide conditional logic when needed</solution> 
    </anti_pattern> 
     
  </anti_patterns> 
   
</philosophical_foundation>

      <name>Chain of Thought (CoT)</name> 
      <description>Force the model to show explicit reasoning steps before reaching

      <effectiveness>Reduces hallucinations by 40-60% on reasoning tasks</effectiveness>

            Before generating your final output, work through these steps: 
             
            STEP 1 - UNDERSTAND: Restate the core problem in your own words 
            STEP 2 - ANALYZE: Identify key variables, constraints, and relationships   
            STEP 3 - PLAN: Outline your approach to solving/addressing the problem 
            STEP 4 - EXECUTE: Work through your plan systematically 
            STEP 5 - VERIFY: Check your work for errors or gaps 
            STEP 6 - SYNTHESIZE: Formulate your final response 
             
            Show your thinking for each step in a <thinking> section before your final output. 
            </reasoning_process> 
            ``` 
          </template>
